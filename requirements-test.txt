# Dependencies required to run `pytest`, and nothing more.
#
# requirements.txt carries the full desktop application: PyQt6, MediaPipe,
# OpenCV, pywin32, ADB tooling. The test suite needs none of it. Installing
# only this file is what proves the suite has no Windows, GPU, or hardware
# dependency -- CI does exactly that on ubuntu-latest.
#
# Floors are the versions this suite is verified against, not arbitrary minima.
# Phase 5 adds requirements.lock with exact pins for reproducible installs.

pytest>=8.3
numpy>=2.0
loguru>=0.7

# requests is not in any module-level import of the code under test --
# src/jarvis/core/tts_engine.py imports it lazily inside a function. It is still
# required: tests/test_tts_engine.py calls
# monkeypatch.setattr("requests.post", ...), and that string form makes pytest
# import requests itself.
requests>=2.32

# Coverage measurement and the CI gate. Kept here rather than installed ad hoc
# so the number CI enforces is reproducible locally with the same command.
pytest-cov>=5.0

# src/jarvis/core/intent_router.py imports yaml at module level, so the router
# characterization tests need it even though they never load a config file.
pyyaml>=6.0

# sounddevice and soundfile load PortAudio / libsndfile at import time, so
# `import jarvis.core.tts_engine` needs the shared libraries present. No audio
# device is ever opened: tests/test_tts_engine.py monkeypatches every playback
# call and every requests.post. On Debian/Ubuntu the libraries are
# `libportaudio2` and `libsndfile1`.
sounddevice>=0.5
soundfile>=0.13

# The linter, replacing flake8. Not a test dependency in the strict sense, but
# it belongs with the tools CI runs so that `ruff check .` locally means what it
# means in CI -- the rule set itself lives in pyproject.toml's [tool.ruff].
#
# The floor is the version the rule set was measured against, 0.12.0, and it is
# a floor with teeth: ruff's default rule set changes between releases, which is
# precisely why pyproject.toml pins select/ignore explicitly instead of relying
# on a default. The CI lint job repeats this floor rather than reading this file
# (it deliberately installs a short package list, not the whole test env); keep
# the two in step.
ruff>=0.12
